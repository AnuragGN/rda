1. Install nfs server

2. Create shared directory to serve
   sudo mkdir /storage
   sudo chown -R nobody:nogroup /storage
   sudo chmod -R 777 /storage

3. Add share to nfs exports
   sudo vi /etc/exports

   Add the following line to /etc/exports.
   /storage *(rw,sync,no_subtree_check)
   /storage 192.168.1.5(rw,sync,no_subtree_check)

4. Restart nfs-kernel-server
   sudo systemctl restart nfs-kernel-server

5. Confirm that the directory is being shared by running
   sudo exportfs

A.2.1 $vi mount.sh
------------------------------------------------------------------------------------------
#!/bin/sh

now=$(date)

point=$1
directory=$2

if mountpoint -q "$directory" ; then
        echo "$now" : "$directory" is mounted. >> /var/log/nfs-mounted.log
else
        echo "$now" : Mounting "$directory" >> /var/log/nfs-mounting.log
        mount "$point" "$directory" >> /var/log/nfs-mounting.log 2>&1
fi
------------------------------------------------------------------------------------------
